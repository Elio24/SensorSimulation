cmake_minimum_required(VERSION 3.20)
project(SimulateurIOT LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(iot
    src/Sensor.cpp src/TemperatureSensor.cpp src/HumiditySensor.cpp
    src/SoundSensor.cpp src/LightSensor.cpp src/Server.cpp src/Scheduler.cpp)
target_include_directories(iot PUBLIC include)
if(MSVC)
    target_compile_options(iot PRIVATE /W4 /utf-8)
else()
    target_compile_options(iot PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(SimulateurIOT src/main.cpp)
target_link_libraries(SimulateurIOT PRIVATE iot)
include(CTest)
if(BUILD_TESTING)
    add_executable(iot_tests tests/test_simulator.cpp)
    target_link_libraries(iot_tests PRIVATE iot)
    add_test(NAME simulator_contracts COMMAND iot_tests)
endif()
